cmake_minimum_required(VERSION 3.22.1)
project(rnnoise_jni LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
set(RNNOISE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/vendor/rnnoise")
if(NOT EXISTS "${RNNOISE_ROOT}/src/rnnoise_data.c")
  message(FATAL_ERROR "RNNoise sources/model missing; execute bash scripts/fetch_rnnoise.sh before Gradle build")
endif()
set(RNNOISE_SOURCES
  denoise.c rnn.c pitch.c kiss_fft.c celt_lpc.c nnet.c nnet_default.c
  parse_lpcnet_weights.c rnnoise_data.c rnnoise_tables.c)
foreach(src IN LISTS RNNOISE_SOURCES)
  list(APPEND FULL_SOURCES "${RNNOISE_ROOT}/src/${src}")
endforeach()
add_library(rnnoise_core STATIC ${FULL_SOURCES})
target_include_directories(rnnoise_core PUBLIC "${RNNOISE_ROOT}/include" "${RNNOISE_ROOT}/src")
target_compile_options(rnnoise_core PRIVATE -O3 -fvisibility=hidden)
add_library(rnnoise_jni SHARED rnnoise_jni.cpp)
target_link_libraries(rnnoise_jni PRIVATE rnnoise_core m log)
